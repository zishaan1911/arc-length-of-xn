\begin{abstract}
The graph of $y=x^n$ on $[0,1]$ has arc length $L(n)\to2$ as $n\to\infty$, because the curve
collapses onto the two legs of a right-angled corner. We determine the rate of this
convergence. Comparing Euclidean with taxicab length and evaluating a single Beta
integral, we derive an exact, rapidly convergent representation of $L(n)$ in terms of the
Gamma function. From it we obtain $2-L(n)=(\ln n+1-\ln2)/(n-1)+O(\ln^2 n/n^2)$, the sharper
approximation $L(n)=1+(2/(en))^{1/(n-1)}+\pi^2/(12n^2)+O(\ln n/n^3)$, and elementary two-sided
bounds valid for all $n\ge3$. The logarithm arises because the curve approaches
the vertical leg of the corner only logarithmically. For a corner of arbitrary turning angle
$\theta$, the coefficient of $\ln n/n$ becomes $2\sin^2(\theta/2)$, and we find the next-order
constant in closed form. By contrast, the perimeter of the superellipse $|x|^p+|y|^p=1$
satisfies $P(p)=8-4K/p+O(p^{-2})$ with $K=2\sqrt2\ln(1+\sqrt2)-2\ln2$, with no logarithm,
because the superellipse approaches both sides of the square exponentially fast. All
numerical results were checked in high-precision arithmetic with publicly available code.
\end{abstract}

\section{Introduction}\label{sec:intro}
% ======================================================================

\IEEEPARstart{L}{ength} is not continuous under uniform convergence. A staircase from
$(0,0)$ to $(1,1)$ with ever finer steps converges uniformly to the diagonal, yet its length
stays equal to $2$ instead of approaching $\sqrt2$, because arc length is only lower
semicontinuous~\cite{BBI}.

The family $y=x^n$ on $[0,1]$ is a smooth relative of the staircase. As $n\to\infty$ the
graphs converge, in the Hausdorff sense, to the corner path
\[
  \Gamma:\ (0,0)\longrightarrow(1,0)\longrightarrow(1,1),
\]
and this time the lengths do converge to the length of the limit:
\begin{equation}\label{eq:Ldef}
  L(n):=\int_0^1\sqrt{1+n^2x^{2n-2}}\dd x\ \longrightarrow\ 2.
\end{equation}
This limit is a standard exercise. In this paper we study the rate of convergence. We show that
the error decays like $(\ln n)/n$, explain the origin of the logarithm, and give an exact
expansion of the error in terms of the Gamma function.
Fig.~\ref{fig:curves} shows the two families of curves studied in this paper.

\subsection{Main Results}\label{sec:main}

Throughout, $n>1$ is real, $\ln$ denotes the natural logarithm, and
\begin{equation}\label{eq:epsD}
  \eps:=\frac{1}{n-1},\qquad D(n):=2-L(n).
\end{equation}
We call $D(n)$ the \emph{defect}. Table~\ref{tab:notation} lists the notation used
throughout. Our first result is elementary and has explicit constants.

\begin{mainthm}[Proposition~\ref{prop:elementary}]\label{thmA}
For every $n\ge3$,
\begin{align*}
  D(n)&\le\frac{\ln n+1}{n-1},\\
  D(n)&\ge\frac{\ln n-1}{n-1}-\frac{\ln^2n}{2(n-1)^2}.
\end{align*}
In particular, $D(n)\sim(\ln n)/n$ as $n\to\infty$.
\end{mainthm}

The second result is exact. For $0\le\eps<1$ let
\begin{equation}\label{eq:Rdef}
  R(\eps):=\frac{\Gamma\bigl(1+\frac{\eps}{2}\bigr)\,\Gamma\bigl(\frac{1-\eps}{2}\bigr)}
  {\sqrt\pi\,(1+\eps)},
\end{equation}
so that $R(0)=1$.

\begin{mainthm}[Theorem~\ref{thm:exact} and Corollary~\ref{cor:asymp}]\label{thmB}
For every real $n>2$,
\begin{multline}\label{eq:exact-intro}
  L(n)=1+\frac{R(\eps)}{n^{\eps}}\\
  -\sum_{k=1}^{\infty}\binom{1/2}{k}\frac{1}{n^{2k-1}\bigl((2k-1)(n-1)-1\bigr)}.
\end{multline}
Consequently, as $n\to\infty$,
\begin{align}
  L(n)&=2-\frac{\ln n+1-\ln2}{n-1}+O\Bigl(\frac{\ln^2n}{n^2}\Bigr),\label{eq:asy1}\\
  L(n)&=1+\Bigl(\frac{2}{en}\Bigr)^{\frac{1}{n-1}}+\frac{\pi^2}{12n^2}
        +O\Bigl(\frac{\ln n}{n^3}\Bigr).\label{eq:asy2}
\end{align}
\end{mainthm}

The terms of the series in~\eqref{eq:exact-intro} decay like $n^{-2k}$, and four terms
give $L(100)$ to about $22$ digits. The error of the approximation~\eqref{eq:asy2} is
below $5\times10^{-6}$ at $n=100$ and below $2\times10^{-11}$ at $n=10^4$
(Section~\ref{sec:asymp}).

Section~\ref{sec:why} traces the logarithm to the slow approach of $x^n$ to the vertical leg
of the corner, and Section~\ref{sec:angle} shows that it does not depend on the corner being a
right angle. For unit vectors $u,v\in\R^2$ that make
an angle $\theta\in[0,\pi)$, consider the curve
\begin{equation}\label{eq:gamma}
  \gamma_n(x)=x\,u+x^n\,v,\qquad 0\le x\le1,
\end{equation}
which collapses onto the two-segment path $0\to u\to u+v$ of length $2$ as $n\to\infty$. The
graph of $x^n$ is the case $\theta=\pi/2$.

\begin{mainthm}[Theorem~\ref{thm:angle}]\label{thmC}
Let $L_\theta(n)$ be the length of $\gamma_n$. Then
\[
  2-L_\theta(n)=\frac{2\sin^2\frac{\theta}{2}\,\ln n+C(\theta)}{n-1}
  +O\Bigl(\frac{\ln^2n}{n^2}\Bigr),
\]
where
\[
  C(\theta)=2\sin^2\frac{\theta}{2}+4\cos^2\frac{\theta}{2}\,\ln\cos\frac{\theta}{2}.
\]
\end{mainthm}

The logarithm does not appear for every curve that collapses onto a corner. The superellipse
$|x|^p+|y|^p=1$, popularized by Piet Hein~\cite{Gardner}, converges to the square $[-1,1]^2$
as $p\to\infty$ at the faster rate $1/p$.

\begin{mainthm}[Theorem~\ref{thm:squircle}]\label{thmD}
The perimeter $P(p)$ of $|x|^p+|y|^p=1$ satisfies, for every $p\ge2$,
\[
  \Bigl|\,P(p)-8+\frac{4K}{p}\Bigr|\le\frac{12}{p^2},
\]
where $K=2\sqrt2\,\ln(1+\sqrt2)-2\ln2=1.10660\ldots$.
\end{mainthm}

\subsection{Related Work}\label{sec:related}

The limit $L(n)\to2$ is standard. The indefinite arc-length integral of $x^n$ is a Gauss
hypergeometric function; see the discussion in~\cite{MSE}, which gives in particular
\begin{equation}\label{eq:hyp}
  L(n)={}_2F_1\Bigl(-\frac12,\frac{\eps}{2};\,1+\frac{\eps}{2};\,-n^2\Bigr).
\end{equation}
In principle, \eqref{eq:exact-intro} can be obtained from~\eqref{eq:hyp} through the classical
connection formula that transforms a ${}_2F_1$ at argument $z$ into functions of
$1/z$~\cite[Eq.~15.8.2]{DLMF}, \cite[Ch.~2]{AAR}; see Remark~\ref{rem:hyp}. Our derivation
does not use hypergeometric functions. It needs one substitution and one Beta integral, and
it leads directly to the asymptotic statements~\eqref{eq:asy1} and~\eqref{eq:asy2}, which we
have not found in the literature.

Two recent preprints~\cite{MRL,RM} give hypergeometric series for the arc length of
supercircles and for the perimeter of Lam\'e superellipses, and they verify the limiting
value $8$ as $p\to\infty$. They do not give the rate $4K/p$ of Theorem~\ref{thmD}, and we have
not found it elsewhere.

The integrals behind our results have the form $\int_0^n h(s)\,s^{\eps-1}\dd s$ with a small
parameter $\eps$, which is the natural setting of Mellin-transform
asymptotics~\cite{FGD,Wong}; see also~\cite{deBruijn} for the classical methods. We need only
one Mellin transform, which we evaluate directly. Because these questions are elementary, some
of our results may already appear in the problem sections of journals, and we would be grateful
for references.

\subsection{Contributions and Organization}\label{sec:contrib}

The contributions of this paper are as follows.
\begin{itemize}
\item A reduction of each problem to a single one-dimensional integral, by comparing

\item Explicit two-sided bounds for the defect $D(n)$ (Section~\ref{sec:elementary}).
\item An exact, rapidly convergent Gamma-function series for $L(n)$, derived from one Beta

\end{itemize}

\section{The Taxicab Defect}\label{sec:defect}
% ======================================================================

We compare the Euclidean length of a curve with its \emph{taxicab} length. Let
$\gamma(t)=(x(t),y(t))$ be a $C^1$ curve from $(0,0)$ to $(1,1)$ whose two coordinates are
monotone. Its taxicab length is $\int(|x'|+|y'|)\dd t=1+1=2$, so
\begin{equation}\label{eq:taxicab}
  2-\operatorname{length}(\gamma)=\int\Bigl(|x'|+|y'|-\sqrt{x'^2+y'^2}\Bigr)\dd t\ \ge0.
\end{equation}
The integrand vanishes where the curve is horizontal or vertical and is largest where the
curve moves diagonally, so the defect measures how far the curve travels in diagonal
directions. The same comparison works for any pair of directions.

\begin{lemma}[Defect integral]\label{lem:defect}
Let $u,v\in\R^2$ be unit vectors with $u\cdot v=c=\cos\theta$, where $\theta\in[0,\pi)$, and
let $\gamma_n$ be as in~\eqref{eq:gamma}. Then
\begin{equation}\label{eq:defect}
  2-\operatorname{length}(\gamma_n)=\eps\,n^{-\eps}\int_0^n h_c(s)\,s^{\eps-1}\dd s,
\end{equation}
where, for $s\ge0$,
\begin{align}
  h_c(s)&:=1+s-\sqrt{1+2cs+s^2}\notag\\
  &\phantom{:}=\frac{2(1-c)\,s}{1+s+\sqrt{1+2cs+s^2}}.\label{eq:hc}
\end{align}
\end{lemma}

\begin{IEEEproof}
Write $y=x^n$. Then $\gamma_n'=u+y'v$ and $|\gamma_n'|=\sqrt{1+2cy'+y'^2}$. Since
$\int_0^1y'\dd x=1$,
\begin{align*}
  2-\operatorname{length}(\gamma_n)&=\int_0^1\bigl(1+y'-|u+y'v|\bigr)\dd x\\
  &=\int_0^1h_c\bigl(nx^{n-1}\bigr)\dd x.
\end{align*}
Now substitute $s=nx^{n-1}$, an increasing bijection from $(0,1]$ onto $(0,n]$, with
$x=(s/n)^{\eps}$ and $\dd x=\eps\,(s/n)^{\eps}\dd s/s$. This gives~\eqref{eq:defect}. The
second expression in~\eqref{eq:hc} follows by multiplying and dividing by
$1+s+\sqrt{1+2cs+s^2}$.
\end{IEEEproof}

The graph of $x^n$ is the case $u=(1,0)$, $v=(0,1)$, $c=0$. We write
\begin{equation}\label{eq:g}
  g(s):=h_0(s)=1+s-\sqrt{1+s^2},
\end{equation}
so that
\begin{equation}\label{eq:D}
  D(n)=\eps\,n^{-\eps}\int_0^n g(s)\,s^{\eps-1}\dd s.
\end{equation}
The function $g$ is the length lost per unit of horizontal distance along a stretch of slope
$s$ (Fig.~\ref{fig:g}). Two properties of $g$ are used below. For small $s$,
$g(s)\approx s$, so gentle slopes lose little length. As $s\to\infty$, $g(s)\to1$, so steep
slopes lose almost the whole horizontal step. The second property makes
$\int_1^n g(s)\,\dd s/s$ grow like $\ln n$.

\section{Elementary Bounds}\label{sec:elementary}
% ======================================================================

\begin{proposition}\label{prop:elementary}
For every real $n\ge3$,
\begin{align*}
  D(n)&\le\frac{\ln n+1}{n-1},\\
  D(n)&\ge\frac{\ln n-1}{n-1}-\frac{\ln^2n}{2(n-1)^2}.
\end{align*}
\end{proposition}

\begin{IEEEproof}
We use two bounds on $g$. Since $1+s+\sqrt{1+s^2}\ge1+s+\max(1,s)\ge2\max(1,s)$, the second
form of $g$ in~\eqref{eq:hc} gives
\begin{equation}\label{eq:gupper}
  0\le g(s)\le\min(s,1).
\end{equation}
Also $g(s)=1-(\sqrt{1+s^2}-s)=1-1/(s+\sqrt{1+s^2})$, hence
\begin{equation}\label{eq:glower}
  g(s)\ge1-\frac{1}{2s}\qquad(s>0).
\end{equation}

\emph{Upper bound.} By~\eqref{eq:gupper},
\begin{align*}
  \int_0^ng(s)s^{\eps-1}\dd s&\le\int_0^1s^{\eps}\dd s+\int_1^ns^{\eps-1}\dd s\\
  &=\frac{1}{1+\eps}+\frac{n^{\eps}-1}{\eps}.
\end{align*}
Multiplying by $\eps n^{-\eps}$ gives $D(n)\le\eps+(1-n^{-\eps})\le\eps+\eps\ln n$, because
$1-e^{-x}\le x$.

\emph{Lower bound.} Drop the nonnegative integral over $[0,1]$ and apply~\eqref{eq:glower}
on $[1,n]$:
\begin{align*}
  D(n)&\ge\eps n^{-\eps}\int_1^n\Bigl(s^{\eps-1}-\frac12s^{\eps-2}\Bigr)\dd s\\
  &=(1-n^{-\eps})-\frac{\eps n^{-\eps}}{2}\cdot\frac{1-n^{\eps-1}}{1-\eps}\\
  &\ge(1-n^{-\eps})-\frac{\eps}{2(1-\eps)}.
\end{align*}
For $n\ge3$ we have $\eps\le\frac12$, so $\eps/(2(1-\eps))\le\eps$. Finally,
$1-e^{-x}\ge x-x^2/2$ for $x\ge0$; with $x=\eps\ln n$ this gives the claim.
\end{IEEEproof}

Proposition~\ref{prop:elementary} already shows that $n\,(2-L(n))/\ln n\to1$. Its bounds leave
the constant next to $\ln n$ undetermined within $[-1,1]$. Section~\ref{sec:asymp} shows that
this constant equals $1-\ln2\approx0.307$.

\section{An Exact Formula}\label{sec:exact}
% ======================================================================

The function $k(s):=\sqrt{1+s^2}-s=1-g(s)$ is positive and decreasing, with $k(0)=1$ and
$k(s)\le1/(2s)$. Its \emph{Mellin transform}
\begin{equation}\label{eq:mellin}
  M(\eps):=\int_0^\infty k(s)\,s^{\eps-1}\dd s
\end{equation}
therefore converges for $0<\eps<1$. Mellin transforms are the standard tool for integrals such
as~\eqref{eq:D}~\cite{FGD,Wong}; here we need only the evaluation in Lemma~\ref{lem:mellin}.

\begin{lemma}\label{lem:mellin}
For $0<\eps<1$, we have $\eps\,M(\eps)=R(\eps)$, with $R$ as in~\eqref{eq:Rdef}.
\end{lemma}

\begin{IEEEproof}
Put $w=k(s)=\sqrt{1+s^2}-s$. As $s$ runs over $(0,\infty)$, $w$ decreases from $1$ to $0$.
Solving gives $s=(1-w^2)/(2w)$ and $\dd s=-\frac{1+w^2}{2w^2}\dd w$. Thus
\begin{align*}
  M(\eps)&=\int_0^1 w\Bigl(\frac{1-w^2}{2w}\Bigr)^{\eps-1}\frac{1+w^2}{2w^2}\dd w\\
  &=2^{-\eps}\int_0^1(1-w^2)^{\eps-1}\,w^{-\eps}\,(1+w^2)\dd w.
\end{align*}
Substituting $w=\sqrt t$ gives Euler Beta integrals
$B(a,b)=\int_0^1t^{a-1}(1-t)^{b-1}\dd t$~\cite[Eq.~5.12.1]{DLMF}:
\begin{align*}
  M(\eps)&=2^{-\eps-1}\int_0^1(1-t)^{\eps-1}
    \bigl(t^{-\frac{1+\eps}{2}}+t^{\frac{1-\eps}{2}}\bigr)\dd t\\
  &=2^{-\eps-1}\Bigl[B\Bigl(\frac{1-\eps}{2},\eps\Bigr)+B\Bigl(\frac{3-\eps}{2},\eps\Bigr)\Bigr].
\end{align*}
Write $B(a,b)=\Gamma(a)\Gamma(b)/\Gamma(a+b)$ and use
$\Gamma\bigl(\frac{3\pm\eps}{2}\bigr)=\frac{1\pm\eps}{2}\Gamma\bigl(\frac{1\pm\eps}{2}\bigr)$.
The bracket becomes
\[
  \frac{\Gamma(\eps)\Gamma(\frac{1-\eps}{2})}{\Gamma(\frac{1+\eps}{2})}
  \Bigl(1+\frac{1-\eps}{1+\eps}\Bigr)
  =\frac{\Gamma(\eps)\Gamma(\frac{1-\eps}{2})}{\Gamma(\frac{3+\eps}{2})}.
\]
Hence $\eps M(\eps)=\Gamma(1+\eps)\Gamma(\frac{1-\eps}{2})\big/
\bigl(2^{1+\eps}\Gamma(\frac{3+\eps}{2})\bigr)$. Legendre's duplication formula
$\Gamma(1+\eps)=\pi^{-1/2}2^{\eps}\Gamma(\frac{1+\eps}{2})\Gamma(1+\frac{\eps}{2})$
\cite[Eq.~5.5.5]{DLMF}, \cite[Ch.~1]{AAR}, together with
$\Gamma(\frac{3+\eps}{2})=\frac{1+\eps}{2}\Gamma(\frac{1+\eps}{2})$, turns this into
$R(\eps)$.
\end{IEEEproof}

\begin{theorem}[Exact formula]\label{thm:exact}
For every real $n>2$, with $\eps=1/(n-1)$,
\begin{multline}\label{eq:exact}
  L(n)=1+\frac{R(\eps)}{n^{\eps}}\\
  -\sum_{k=1}^{\infty}\binom{1/2}{k}\frac{1}{n^{2k-1}\bigl((2k-1)(n-1)-1\bigr)},
\end{multline}
where $\binom{1/2}{k}=\frac{1}{k!}\prod_{j=0}^{k-1}\bigl(\frac12-j\bigr)$. The coefficients are
$\frac12,-\frac18,\frac1{16},-\frac5{128},\dots$, and the series converges absolutely.
\end{theorem}

\begin{IEEEproof}
Since $n>2$, we have $0<\eps<1$. By~\eqref{eq:D} and $g=1-k$,
\begin{align*}
  \int_0^ng(s)s^{\eps-1}\dd s&=\frac{n^{\eps}}{\eps}-\int_0^\infty k(s)s^{\eps-1}\dd s\\
  &\quad+\int_n^\infty k(s)s^{\eps-1}\dd s.
\end{align*}
Multiplying by $\eps n^{-\eps}$ and applying Lemma~\ref{lem:mellin} gives
\begin{equation}\label{eq:Dexact}
  D(n)=1-\frac{R(\eps)}{n^{\eps}}+\eps\,n^{-\eps}\int_n^\infty k(s)\,s^{\eps-1}\dd s.
\end{equation}
For $s>1$, the binomial series gives
\[
  k(s)=s\bigl(\sqrt{1+s^{-2}}-1\bigr)=\sum_{k\ge1}\binom{1/2}{k}s^{1-2k}.
\]
Because $\sum_k|\binom{1/2}{k}|\,n^{-2k}<\infty$ for $n>1$, we may integrate term by term on
$[n,\infty)$ by Tonelli's theorem. Each term contributes
$\int_n^\infty s^{\eps-2k}\dd s=n^{\eps+1-2k}/(2k-1-\eps)$. Hence
\begin{align*}
  &\eps\,n^{-\eps}\int_n^\infty k(s)s^{\eps-1}\dd s\\
  &\qquad=\sum_{k\ge1}\binom{1/2}{k}\frac{\eps\,n^{1-2k}}{2k-1-\eps}\\
  &\qquad=\sum_{k\ge1}\binom{1/2}{k}\frac{1}{n^{2k-1}\bigl((2k-1)(n-1)-1\bigr)}.
\end{align*}
Finally, $L(n)=2-D(n)$.
\end{IEEEproof}

\begin{example}[$n=3$]\label{ex:n3}
Here $\eps=\frac12$ and $R(\frac12)=\Gamma(\frac54)\Gamma(\frac14)/(\frac32\sqrt\pi)
=\Gamma(\frac14)^2/(6\sqrt\pi)$. Theorem~\ref{thm:exact} gives the length of $y=x^3$ on
$[0,1]$ as
\[
  L(3)=1+\frac{\Gamma(\frac14)^2}{6\sqrt{3\pi}}-\sum_{k\ge1}\binom{1/2}{k}\frac{1}{3^{2k-1}(4k-3)}
  =1.547865\ldots,
\]
in agreement with direct quadrature of $\int_0^1\sqrt{1+9x^4}\dd x$ to 40 digits. The
Gamma-function term equals $0.713634\ldots$, and the series converges geometrically with
ratio $1/9$.
\end{example}

\begin{remark}[The case $n=2$]\label{rem:n2}
At $n=2$ we have $\eps=1$. Both $R(\eps)$, through $\Gamma(\frac{1-\eps}{2})$, and the $k=1$
term of the series have poles, and the two poles cancel. Since $L$ is continuous in $n$, the
limit $n\to2^+$ of~\eqref{eq:exact} is the length of the parabola,
$L(2)=\frac{\sqrt5}{2}+\frac14\ln(2+\sqrt5)$.
\end{remark}

\begin{remark}[Scaled curves]\label{rem:scaled}
The same proof applies to $y=c\,x^n$ on $[0,1]$ for any $c>0$ with $cn>1$; the only change is
that the substitution now ends at $s=cn$. Writing $\ell_c(n)$ for its length,
\begin{multline}\label{eq:scaled}
  \ell_c(n)=c+\frac{R(\eps)}{(cn)^{\eps}}\\
  -\sum_{k\ge1}\binom{1/2}{k}\frac{1}{(cn)^{2k-1}\bigl((2k-1)(n-1)-1\bigr)}.
\end{multline}
Scaling shows that the length of $y=x^n$ on $[0,a]$ equals $a\,\ell_{a^{n-1}}(n)$, so this also
covers every interval $[0,a]$.
\end{remark}

\begin{remark}[Relation to the hypergeometric form]\label{rem:hyp}
The large-argument connection formula~\cite[Eq.~15.8.2]{DLMF} expresses~\eqref{eq:hyp} as a
combination of two ${}_2F_1$'s at argument $-1/n^2$. One can check that the Gamma factor
multiplying $n^{-\eps}$ there, $\Gamma(1+\frac\eps2)\Gamma(-\frac{1+\eps}{2})/\Gamma(-\frac12)$,
equals $R(\eps)$. The derivation given here is elementary, and the form~\eqref{eq:exact}
leads directly to the asymptotic results of the next section.
\end{remark}

\section{Asymptotic Expansion}\label{sec:asymp}
% ======================================================================

We need the Taylor expansion of $\ln R$ at $0$. It uses the classical values
$\psi(1)=-\gamma$, $\psi(\frac12)=-\gamma-2\ln2$, $\psi'(1)=\pi^2/6$, $\psi'(\frac12)=\pi^2/2$,
$\psi''(1)=-2\zeta(3)$, and $\psi''(\frac12)=-14\zeta(3)$, where $\psi=\Gamma'/\Gamma$ and
$\gamma$ is Euler's constant~\cite[Secs.~5.4 and~5.15]{DLMF}.

\begin{lemma}\label{lem:logR}
As $\eps\to0$,
\[
  \ln R(\eps)=a_1\eps+a_2\eps^2+a_3\eps^3+O(\eps^4),
\]
where $a_1=\ln2-1$, $a_2=\frac{\pi^2}{12}+\frac12$, and $a_3=\frac{\zeta(3)}{4}-\frac13$.
\end{lemma}

\begin{IEEEproof}
We have $\ln R(\eps)=\ln\Gamma(1+\frac\eps2)+\ln\Gamma(\frac{1-\eps}{2})-\frac12\ln\pi
-\ln(1+\eps)$, and $R$ is analytic near $0$. The $j$th Taylor coefficient of
$\ln\Gamma(\alpha+\beta\eps)$ is $\beta^j\psi^{(j-1)}(\alpha)/j!$. Hence
\begin{align*}
  a_1&=\tfrac12\psi(1)-\tfrac12\psi(\tfrac12)-1=\ln2-1,\\
  a_2&=\tfrac18\psi'(1)+\tfrac18\psi'(\tfrac12)+\tfrac12=\tfrac{\pi^2}{48}+\tfrac{\pi^2}{16}+\tfrac12,\\
  a_3&=\tfrac1{48}\psi''(1)-\tfrac1{48}\psi''(\tfrac12)-\tfrac13=\tfrac{12\zeta(3)}{48}-\tfrac13.
\end{align*}
Euler's constant cancels in $a_1$.
\end{IEEEproof}

\begin{corollary}[Asymptotics of $L(n)$]\label{cor:asymp}
As $n\to\infty$,
\begin{align*}
  &\text{(a)}\ \ L(n)=2-\frac{\ln n+1-\ln2}{n-1}+O\Bigl(\frac{\ln^2n}{n^2}\Bigr),\\
  &\text{(b)}\ \ L(n)=1+\Bigl(\frac{2}{en}\Bigr)^{\frac{1}{n-1}}+\frac{\pi^2}{12n^2}
    +O\Bigl(\frac{\ln n}{n^3}\Bigr).
\end{align*}
\end{corollary}

\begin{IEEEproof}
Put $\lambda=\ln n+1-\ln2=\ln n-a_1$. By Lemma~\ref{lem:logR},
\begin{align*}
  \frac{R(\eps)}{n^{\eps}}&=\exp\bigl(-\eps\lambda+a_2\eps^2+O(\eps^3)\bigr)\\
  &=e^{-\eps\lambda}\bigl(1+a_2\eps^2+O(\eps^3)\bigr).
\end{align*}
The series in Theorem~\ref{thm:exact} equals $\frac{1}{2n(n-2)}+O(n^{-4})$.

For (a), note that $e^{-\eps\lambda}=1-\eps\lambda+O(\eps^2\ln^2n)$, and that every other term
is $O(n^{-2})$.

For (b), note that $e^{-\eps\lambda}=1+O(\eps\ln n)$, so
$e^{-\eps\lambda}a_2\eps^2=a_2\eps^2+O(\ln n/n^3)$. Hence
\[
  L(n)=1+e^{-\eps\lambda}+a_2\eps^2-\frac{1}{2n(n-2)}+O\Bigl(\frac{\ln n}{n^3}\Bigr).
\]
Now $a_2\eps^2=a_2/n^2+O(n^{-3})$, $\frac{1}{2n(n-2)}=\frac{1}{2n^2}+O(n^{-3})$, and
$a_2-\frac12=\frac{\pi^2}{12}$. Finally,
$e^{-\eps\lambda}=\exp\bigl(-\frac{\ln n+1-\ln2}{n-1}\bigr)=\bigl(\frac{2}{en}\bigr)^{1/(n-1)}$.
\end{IEEEproof}

The same argument, carried one step further, yields the next two terms
(Proposition~\ref{prop:higher} in Appendix~\ref{app:higher}). All further terms can be computed
in the same way; they are polynomials in $\ln n$ divided by powers of $n$.

\begin{corollary}[Scaled curves]\label{cor:scaled}
Fix $c>0$, and let $\ell_c(n)$ be the length of $y=c\,x^n$ on $[0,1]$, which tends to $1+c$.
As $n\to\infty$,
\[
  1+c-\ell_c(n)=\frac{\ln(cn)+1-\ln2}{n-1}+O\Bigl(\frac{\ln^2n}{n^2}\Bigr).
\]
\end{corollary}

\begin{IEEEproof}
By~\eqref{eq:scaled}, $1+c-\ell_c(n)=1-(cn)^{-\eps}R(\eps)+S_c(n)$, where the series $S_c(n)$
equals $\frac{1}{2cn(n-2)}+O(n^{-4})=O(n^{-2})$ for fixed $c$. By Lemma~\ref{lem:logR},
$(cn)^{-\eps}R(\eps)=\exp\bigl(-\eps(\ln(cn)+1-\ln2)+O(\eps^2)\bigr)$, which equals
$1-\eps(\ln(cn)+1-\ln2)+O(\eps^2\ln^2n)$.
\end{IEEEproof}

Stretching the vertical leg of the corner by a factor $c$ therefore changes the defect only
through $\ln c$, by $(\ln c)/(n-1)$ to first order.

\section{Corners of Arbitrary Angle}\label{sec:angle}
% ======================================================================

The leading term in Corollary~\ref{cor:asymp}(a) depends only on two properties of $g$: it is
roughly linear near $0$, and it has a limit at $\infty$. Lemma~\ref{lem:general} treats any
function with these two properties.

\begin{lemma}\label{lem:general}
Let $h:[0,\infty)\to\R$ be continuous. Suppose there are constants $A,B\ge0$ and $h_\infty$
such that $|h(s)|\le As$ for $0\le s\le1$ and $|h(s)-h_\infty|\le B/s$ for $s\ge1$. Put
\[
  C_h:=\int_0^1\frac{h(s)}{s}\dd s+\int_1^\infty\frac{h(s)-h_\infty}{s}\dd s.
\]
Then for $n\ge3$ and $\eps=1/(n-1)$,
\begin{multline*}
  \eps\,n^{-\eps}\int_0^nh(s)\,s^{\eps-1}\dd s\\
  =\eps\bigl(h_\infty\ln n+C_h\bigr)+O(\eps^2\ln^2n),
\end{multline*}
where the implied constant depends only on $A$, $B$, and $h_\infty$.
\end{lemma}

\begin{IEEEproof}
Split $J:=\int_0^nh(s)s^{\eps-1}\dd s$ as
\begin{align*}
  J&=\int_0^1h(s)s^{\eps-1}\dd s+h_\infty\int_1^ns^{\eps-1}\dd s\\
  &\quad+\int_1^n\bigl(h(s)-h_\infty\bigr)s^{\eps-1}\dd s.
\end{align*}
We compare each piece with its value at $\eps=0$. For $0<s\le1$ we have
$|s^{\eps}-1|\le\eps|\ln s|$, so the first piece differs from $\int_0^1h(s)\dd s/s$ by at most
$A\eps\int_0^1|\ln s|\dd s=A\eps$. For $1\le s\le n$ we have
$|s^{\eps}-1|\le\eps\ln s\cdot n^{\eps}\le e\,\eps\ln s$, because
$n^{\eps}=e^{\ln n/(n-1)}\le e$. So the third piece differs from
$\int_1^\infty(h-h_\infty)\dd s/s$ by at most
$eB\eps\int_1^\infty s^{-2}\ln s\dd s+\int_n^\infty Bs^{-2}\dd s\le eB\eps+B/n$. Therefore
\[
  J=h_\infty\frac{n^{\eps}-1}{\eps}+C_h+E,\qquad|E|\le(A+eB+B)\,\eps.
\]
Multiplying by $\eps n^{-\eps}$ gives $h_\infty(1-n^{-\eps})+\eps n^{-\eps}(C_h+E)$. Put
$x=\eps\ln n\in(0,1]$. Then $1-e^{-x}=x+O(x^2)$ and $e^{-x}=1+O(x)$, and since $\ln n\ge1$,
every error term is $O(\eps^2\ln^2n)$.
\end{IEEEproof}

\begin{theorem}[Corners of arbitrary angle]\label{thm:angle}
Let $u,v$ be unit vectors at angle $\theta\in[0,\pi)$, and let $L_\theta(n)$ be the length of
$\gamma_n(x)=x\,u+x^n\,v$, $0\le x\le1$. Then
\[
  2-L_\theta(n)=\frac{2\sin^2\frac{\theta}{2}\,\ln n+C(\theta)}{n-1}
  +O\Bigl(\frac{\ln^2n}{n^2}\Bigr),
\]
where
\begin{equation}\label{eq:Ctheta}
  C(\theta)=2\sin^2\frac{\theta}{2}+4\cos^2\frac{\theta}{2}\,\ln\cos\frac{\theta}{2}.
\end{equation}
\end{theorem}

For $\theta=\pi/2$ this gives $C=1-\ln2$, in agreement with Corollary~\ref{cor:asymp}(a). For
$\theta=0$ both terms vanish, since $\gamma_n$ is then a straight segment. The
coefficient $2\sin^2(\theta/2)=1-\cos\theta$ increases with the turning angle.

\begin{IEEEproof}
By Lemma~\ref{lem:defect}, we must apply Lemma~\ref{lem:general} to $h=h_c$ with
$c=\cos\theta\in(-1,1]$. Write $r(s)=\sqrt{1+2cs+s^2}$. The second form of~\eqref{eq:hc} gives
$0\le h_c(s)\le(1-c)s$, so $A=1-c$ works. A short computation gives
\[
  h_c(s)-(1-c)=\frac{(1-c)\bigl(s-1-r(s)\bigr)}{1+s+r(s)}.
\]
Since $|s+c|\le r(s)\le s+1$, we have $|s-1-r(s)|\le2$. Therefore $h_\infty=1-c$ and
$B=2(1-c)$ work.

It remains to compute $C_{h_c}$. Differentiation confirms that
\[
  G(s)=r(s)+c\ln\bigl(s+c+r(s)\bigr)-\ln\frac{1+cs+r(s)}{s}
\]
satisfies $G'(s)=r(s)/s$. (Since $r(s)\ge|s+c|$ and $r(s)\ge|1+cs|$, with $c>-1$, both
logarithms are defined for $s>0$.) Hence
$\int_\delta^Nh_c(s)\,\dd s/s=\bigl[\ln s+s-G(s)\bigr]_\delta^N$, and
\[
  C_{h_c}=\lim_{\substack{\delta\to0\\N\to\infty}}
  \Bigl(\int_\delta^N\frac{h_c(s)}{s}\dd s-(1-c)\ln N\Bigr).
\]
As $N\to\infty$, we have $r(N)=N+c+O(1/N)$, $N+c+r(N)\sim2N$, and $(1+cN+r(N))/N\to1+c$. Thus
\[
  \ln N+N-G(N)=(1-c)\ln N-c-c\ln2+\ln(1+c)+o(1).
\]
As $\delta\to0$, we have $r(\delta)\to1$ and $(1+c\delta+r(\delta))/\delta\sim2/\delta$. Thus
\[
  \ln\delta+\delta-G(\delta)=-1-c\ln(1+c)+\ln2+o(1).
\]
Subtracting gives
\[
  C_{h_c}=(1-c)+(1+c)\ln\frac{1+c}{2}.
\]
Finally, $1-c=2\sin^2\frac\theta2$ and $\frac{1+c}{2}=\cos^2\frac\theta2$.
\end{IEEEproof}

Fig.~\ref{fig:angle}(a) shows the curves $\gamma_n$ for $\theta=\pi/3$. To test

\section{A Corner Without a Logarithm: The Superellipse}\label{sec:squircle}
% ======================================================================

Let $Q(p)$ be the length of the arc $x^p+y^p=1$, $x,y\ge0$, and let $P(p)=4Q(p)$ be the
perimeter of $|x|^p+|y|^p=1$. At $p=2$, $P=2\pi$.

\begin{theorem}[Superellipse perimeter]\label{thm:squircle}
Let $K:=2\sqrt2\,\ln(1+\sqrt2)-2\ln2=1.1066065994\ldots$. For every $p\ge2$,
\begin{align*}
  \Bigl|\,Q(p)-2+\frac{K}{p}\Bigr|&\le\frac{3}{p^2},\\
  \Bigl|\,P(p)-8+\frac{4K}{p}\Bigr|&\le\frac{12}{p^2}.
\end{align*}
\end{theorem}

\begin{IEEEproof}
Parametrize the arc by $t\in(0,1)$ via $x=t^{1/p}$ and $y=(1-t)^{1/p}$. Put
$\eps=1/p\le\frac12$, $a=t^{\eps-1}$, and $b=(1-t)^{\eps-1}$. Then $|x'|=a/p$ and $|y'|=b/p$,
and by~\eqref{eq:taxicab},
\[
  2-Q(p)=\frac1p\,F(\eps),\qquad F(\eps):=\int_0^1\Phi_\eps(t)\dd t,
\]
where $\Phi_\eps:=a+b-\sqrt{a^2+b^2}$.

\emph{Step 1: $F(0)=K$.} At $\eps=0$, $a=1/t$ and $b=1/(1-t)$. Put $q=\sqrt{t^2+(1-t)^2}$.
Then $\Phi_0=(1-q)/(t(1-t))$, and $1-q=(1-q^2)/(1+q)=2t(1-t)/(1+q)$, so the integrand simplifies
to $\Phi_0(t)=2/(1+q)$. Substitute $w=2t-1$, so that $q^2=(1+w^2)/2$, and then $w=\sinh\phi$.
With $A=\operatorname{arsinh}1=\ln(1+\sqrt2)$,
\begin{align*}
  F(0)&=2\int_0^1\frac{\dd w}{1+\sqrt{(1+w^2)/2}}
       =2\sqrt2\int_0^A\frac{\cosh\phi}{\sqrt2+\cosh\phi}\dd\phi\\
      &=2\sqrt2\,A-4\int_0^A\frac{\dd\phi}{\sqrt2+\cosh\phi}.
\end{align*}
With $\tau=\tanh(\phi/2)$ and $\alpha=\sqrt2+1$, and using $\tanh(A/2)=\sinh A/(1+\cosh A)
=\sqrt2-1$, the last integral becomes
\[
  \int_0^{\sqrt2-1}\frac{2\dd\tau}{(\sqrt2+1)-(\sqrt2-1)\tau^2}
  =\ln\frac{\alpha+\tau}{\alpha-\tau}\Big|_{\tau=\sqrt2-1}=\frac12\ln2.
\]
Hence $F(0)=K$.

\emph{Step 2: $|F(\eps)-K|\le3\eps$.} Since $\Phi_\eps(t)=\Phi_\eps(1-t)$, it suffices to
consider $0<t\le\frac12$. Write $r=\sqrt{a^2+b^2}$. Then $\partial\Phi/\partial a=1-a/r\in[0,1)$,
and likewise for $b$. Also $\partial_\eps a=a\ln t$ and $\partial_\eps b=b\ln(1-t)$. For
$t\le\frac12$ and $\eps\le\frac12$ we have $1\le b\le2$ and $a\ge t^{-1/2}$. Moreover
$(1-a/r)\,a=ab^2/\bigl(r(r+a)\bigr)\le b^2/(2a)$. Hence
\begin{align*}
  |\partial_\eps\Phi_\eps(t)|&\le\frac{b^2}{2a}|\ln t|+b|\ln(1-t)|\\
  &\le2\sqrt t\,|\ln t|+2\ln2\le\frac4e+2\ln2<3.
\end{align*}
By the mean value theorem, $|\Phi_\eps(t)-\Phi_0(t)|\le3\eps$ for every $t$. Integrating over
$[0,1]$ gives $|F(\eps)-K|\le3\eps$. The result follows from $Q(p)=2-F(1/p)/p$.
\end{IEEEproof}

Numerical evaluation of $F'(0)=\int_0^1\partial_\eps\Phi_\eps(t)|_{\eps=0}\dd t$ gives
$F'(0)=-0.5052381924\ldots$. This indicates that
\begin{equation}\label{eq:kappa}
  P(p)=8-\frac{4K}{p}+\frac{\kappa}{p^2}+o(p^{-2}),
\end{equation}
with $\kappa=-4F'(0)=2.0209527696\ldots$, well within the bound of
Theorem~\ref{thm:squircle}. Let $\Delta(p):=P(p)-8+4K/p$.
Table~\ref{tab:sq} and Fig.~\ref{fig:squircle}(b) show $p^2\Delta(p)$ approaching $\kappa$.

Fig.~\ref{fig:squircle}(a) illustrates the origin of $K$. Near the corner $(1,1)$, put
$x=1-\alpha/p$ and $y=1-\beta/p$. To leading order, the equation $x^p+y^p=1$ becomes
$e^{-\alpha}+e^{-\beta}=1$, a single fixed curve that does not depend on~$p$. The constant $K$
is exactly the taxicab defect of this curve; parametrizing it by $e^{-\alpha}=t$ reproduces the
integral for $F(0)$.

\section{The Origin of the Logarithm}\label{sec:why}
% ======================================================================

Formula~\eqref{eq:taxicab} says that length is lost only while the curve moves in a direction
strictly between horizontal and vertical. A steep piece of the curve, with slope $s\gg1$, loses
the fraction $g(s)\approx1$ of its horizontal extent. The defect is therefore approximately the
horizontal distance traveled while the curve is steep.

For $y=x^n$, the slope exceeds $1$ exactly when $x\ge n^{-1/(n-1)}$, an interval of length
\[
  1-n^{-1/(n-1)}=\frac{\ln n}{n}+O\Bigl(\frac{\ln^2n}{n^2}\Bigr).
\]
Near $x=1$ we have $x^n\approx e^{-n(1-x)}$. To climb from height $1/n$, where the slope is
about $1$, to height $1$, the curve must advance horizontally by about $\ln(n)/n$
(Fig.~\ref{fig:steep}). The factor $\ln n$ is the logarithm of the ratio between the
heights $1$ and $1/n$ at which the steep part ends and begins. Equivalently, $x^n$ approaches
the vertical leg $x=1$ only logarithmically, since at height $y$ its distance to the leg is
about $\ln(1/y)/n$.

The superellipse behaves differently. Its steep part is confined to a neighborhood of size
$O(1/p)$ around the corner $(1,1)$, where, as shown in Section~\ref{sec:squircle}, it looks like
the fixed curve $e^{-\alpha}+e^{-\beta}=1$ blown down by the factor $1/p$. That curve approaches
its two asymptotes exponentially fast, so its total diagonal travel is finite, and the
defect is $K/p$ per corner.

To summarize, the defect is of order $1/n$ times the logarithm of the ratio between the largest
and smallest heights at which the curve is steep. This ratio is bounded for the superellipse and
is about $n$ for $x^n$. The same reasoning accounts for the factor $1-\cos\theta$ in
Theorem~\ref{thm:angle}, where only that fraction of each step is lost, and for the term
$\ln c$ in Corollary~\ref{cor:scaled}, where stretching the vertical leg by $c$ multiplies
the ratio of heights by $c$.

\section{Conclusion and Open Problems}\label{sec:conclusion}
% ======================================================================

When a family of curves collapses onto a corner, the length defect measures how far the curves
travel in diagonal directions. For the graph of $x^n$, the defect is $(\ln n+1-\ln2)/(n-1)$ to
first order. We derived this, and the complete asymptotic expansion, from an exact
Gamma-function formula obtained from a single Beta integral. The logarithm persists for every
turning angle, with coefficient $2\sin^2(\theta/2)$, and it is absent for the superellipse,
whose perimeter defect is $4K/p$. The difference lies in how fast each curve approaches the
sides of the corner, which is logarithmic for $x^n$ and exponential for the superellipse. We
close with four open problems.

\begin{problem}
Is there a closed form for the second superellipse constant $\kappa=2.02095276965\ldots$
in~\eqref{eq:kappa}? It equals $-4F'(0)$, an integral of logarithms against algebraic
functions, and we expect dilogarithms to appear. An integer-relation search against $\pi^2$,
$\ln^22$, $\ln^2(1+\sqrt2)$, $\sqrt2\ln2\ln(1+\sqrt2)$, $\ln2$, and $\sqrt2\ln(1+\sqrt2)$ found no
relation.
\end{problem}

\begin{problem}
Theorem~\ref{thm:exact} rests on the Mellin transform of $g=h_0$. Find the analogous exact
formula for general angles, based on the Mellin transform of $h_c$ in~\eqref{eq:hc}. We expect
Legendre functions of $\cos\theta$ to appear.
\end{problem}

\begin{problem}
The heuristic of Section~\ref{sec:why} suggests the following. Let $f:[0,1]\to[0,1]$ be smooth
and increasing, with $f(0)=0$, $f(1)=1$, and $f'(1)>0$. Then the graph of $f^n$ should satisfy
$2-\operatorname{length}(f^n)\sim\ln n/\bigl(f'(1)\,n\bigr)$. Prove this and find the next term.
\end{problem}

\begin{problem}
What happens in higher dimensions? For example, how does the area of the graph of $z=x^ny^n$
over $[0,1]^2$ approach its limit?
\end{problem}

\section{Higher-Order Terms}\label{app:higher}

\begin{proposition}\label{prop:higher}
With $a_2=\frac{\pi^2}{12}+\frac12$ and $d=a_2(1+\ln2)+\frac{\zeta(3)}{4}-\frac43
=1.2063122\ldots$, as $n\to\infty$,
\begin{multline*}
  L(n)=1+\Bigl(\frac{2}{en}\Bigr)^{\frac{1}{n-1}}+\frac{\pi^2}{12n^2}
  +\frac{d-a_2\ln n}{n^3}\\
  +O\Bigl(\frac{\ln^2n}{n^4}\Bigr).
\end{multline*}
\end{proposition}

\begin{IEEEproof}
Keep $\lambda=\ln n-a_1$ as in the proof of Corollary~\ref{cor:asymp}. By
Lemma~\ref{lem:logR},
\[
  \frac{R(\eps)}{n^{\eps}}=e^{-\eps\lambda}\bigl(1+a_2\eps^2+a_3\eps^3+O(\eps^4)\bigr).
\]
Since $e^{-\eps\lambda}=1-\eps\lambda+O(\eps^2\ln^2n)$, the product of $e^{-\eps\lambda}$ with
the bracket equals
\[
  e^{-\eps\lambda}+a_2\eps^2-a_2\eps^3\lambda+a_3\eps^3+O(\eps^4\ln^2n).
\]
The series in Theorem~\ref{thm:exact} equals $\frac{1}{2n(n-2)}+O(n^{-4})
=\frac{1}{2n^2}+\frac{1}{n^3}+O(n^{-4})$, and $\eps^2=\frac{1}{n^2}+\frac{2}{n^3}+O(n^{-4})$,
$\eps^3=\frac{1}{n^3}+O(n^{-4})$. Collecting terms, the coefficient of $n^{-2}$ is
$a_2-\frac12=\frac{\pi^2}{12}$, and the coefficient of $n^{-3}$ is
\[
  2a_2-a_2(\ln n+1-\ln2)+a_3-1=d-a_2\ln n,
\]
because $a_3-1=\frac{\zeta(3)}{4}-\frac43$.
\end{IEEEproof}

\section{Numerical Methods}\label{app:numerics}

All numerical values in this paper were computed in Python with the arbitrary-precision library
\texttt{mpmath}, using 30 to 50 significant digits. The script \texttt{verify.py} in the
accompanying repository recomputes every value and checks every claim listed below; all checks
pass.
\begin{itemize}
